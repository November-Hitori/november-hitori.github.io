bitXor

A⊕B=(A∪B)−(A∩B)=(A∪B)∩(∼(A∩B))=[∼(∼A∩∼B)](∼(A∩B))A\oplus B=(A\cup B)-(A\cap B)=(A\cup B)\cap (\sim(A\cap B))=[\sim(\sim A\cap\sim B)](\sim(A\cap B))A⊕B=(A∪B)−(A∩B)=(A∪B)∩(∼(A∩B))=[∼(∼A∩∼B)](∼(A∩B))

其中最后一步应用了 De-Morgen 律，共 777 步。

int bitXor(int x, int y) {
    return (~((~x)&(~y)))&(~(x&y));
}

leastBitPos

熟悉树状数组的同学都知道，这就是一个 lowbit 操作。
lowbit 的写法是，首先取反，这样原先的 lowbit 位（记为 iii）后面会变成 iii 个 111，而 iii 本身是 000，加 111 后，第 iii 位及往后会全部与原数相同，而取反后前面的都是完全不同的，& 一次即可，共 333 步。

int leastBitPos(int x) {
	return x&((~x)+1);
}

getByte

一个 Byte 是 888 个 Bit，所以左移 8n8n8n 位之后就可以将第 nnn 个 Byte 放在最低的位置上，再 & 上一个全一 Byte 即可，共 333 步。

int getByte(int x, int n) {
    return (x>>(n<<3))&255;
}

logicalShift

直接左移 nnn 位会被首位为 111 卡死，所以考虑去掉这些影响，只需要构造后 32−n32-n32−n 位为 111 的数就行，构造的步骤是这样的：

后 32-n 位为 1
前 n 位为 1（取反）
第一位为 1（左移 n-1 位）

对于左移 n−1n-1n−1 位只需要左移 nnn 位再右移 111 位即可，共 666 步。

int logicalShift(int x, int n) {
    return (x>>n)&(~(((1<<31)>>n)<<1));
}

grayToBinary

很有趣。
考虑倍增，具体而言，我们希望在某一刻，aia_iai​ 存储的是 [i,i+2k−1][i,i+2^{k}-1][i,i+2k−1] 的异或和。
在 k+1k+1k+1 的时候，只需要让 aia_iai​ 异或上 ai+2ka_{i+2^{k}}ai+2k​ 即可，而统一并行操作只需要 << 2^k，共 101010 步。

int grayToBinary(int x) {
    x^=(x>>1); x^=(x>>2);
    x^=(x>>4); x^=(x>>8);
    x^=(x>>16);
    return x;
}

bitCount

有点太难了，兄弟。
考虑分治，每次考虑将高的区间平移到低的位置，和只有低的直接相加（进位不会影响到其他同层区间），这样子每一段分治的过程是全部相同的，于是可以直接并行。
并行只需要构造一些二进制数就行，沟槽的 lab 还不准直接抄常数，这里直接暴力拆还会被卡一下，需要倍增，363636 步。

int bitCount(int x) {
    int r1=(0x55<<8)+0x55;
    int r2=(0x33<<8)+0x33;
    int r3=(0x0F<<8)+0x0F;
    int r4=(0xFF<<16)+0xFF;
    int r5=(0xFF<<8)+0xFF;
    r1=r1|(r1<<16); r2=r2|(r2<<16); r3=r3|(r3<<16);
    x=(x&r1)+((x>>1)&(r1));
    x=(x&r2)+((x>>2)&(r2));
    x=(x&r3)+((x>>4)&(r3));
    x=(x&r4)+((x>>8)&(r4));
    x=(x&r5)+((x>>16)&(r5));
    return x;
}

isEqual

众所周知 x⊕x=0x\oplus x=0x⊕x=0，于是只需要 x^y 再用 ! 判断一下就行，222 步。

int isEqual(int x, int y) {
    int t=x^y;
    return !t;
}

divpwr2

直接左移只有负数可能会炸，判断一下负数（x>>31&1 取符号位）然后判断一下是否整除即可。
是否整除相当于底层 nnn 位有东西，构造一下 2n−12^n-12n−1 and 一下就可以判断，121212 步。

int divpwr2(int x, int n) {
    int rt=(1<<n)+(~0);
    return (x>>n)+((x>>31&1)&(!!(x&rt)));
}

sign

我已经完全学会了 !!x .jpg
左移 313131 位可以取出符号位，此时负数变为 −1-1−1，正数和 000 变为 000，只需要给正数加一。
分别用符号位为 000 和不为 000 判断即可，这些都可以写成位运算的形式，888 步。

int sign(int x) {
    return (x>>31)+((!(x>>31&1))&(!!x));
}

addOK

这个怎么在第三章讲了，好像。
无非是两种情况，两正加起来为负，两负加起来为正，取出符号位判断即可，构造一个比较好的位逻辑即可，111111 步。

int addOK(int x, int y) {
    int t,a,b,c;
    t=x+y,a=(x>>31)&1,b=(y>>31)&1,c=(t>>31)&1;
    return !((a^c)&(b^c));
}

absVal

哎，诡异构造。
考虑必然要先取出符号位 mmm，负数的时候希望得到 −x-x−x，而 m=−1m=-1m=−1，考虑到 ~x+1=-x，而 ~x= x ^ -1，111 的话可以 m&1 构造出来，于是可以写出 (x^m)+(m&1)，m=0m=0m=0 也即正数的时候，当然也符合了条件，444 步。

int absVal(int x) {
    int t=x>>31;
    return (x^t)+(t&1);
}

satSub

从 addOK 出发可以知道溢出大概是有两种情况的，不过这回两种情况要分开判断，取出 x,y,x−yx,y,x-yx,y,x−y 的符号位，然后考虑怎么样来分开判断。
考虑利用 0xff and 上一个数来保留，0 and 上一个数来删除，于是只需要构造 Tmax⁡,Tmin⁡T_{\max},T_{\min}Tmax​,Tmin​ 以及三个 0xff 或者 0 就行，242424 步。

int satSub(int x, int y) {
    int t,sx,sy,st,Tmax,Tmin,b1,b2,b3;
    t=x+(~y)+1;
    sx=(x>>31),sy=(y>>31),st=(t>>31);
    Tmax=~(1<<31),Tmin=(1<<31);
    b1=(~sx)&sy&st,b2=sx&(~sy)&(~st);
    b3=b1^b2^(~0);
    return (b1&Tmax)+(b2&Tmin)+(b3&t);
}

float_twice

解禁了常数！！！终于不用自己拼了哈哈。
分讨题，首先取出阶码：

  • 如果阶码是 255255255，则返回原值即可。
  • 如果阶码大于零，说明是规格化数，直接阶码 +1+1+1，
    • 需要额外讨论阶码是 254254254，×2\times 2×2 后得到 inf，于是需要处理掉所有的小数位，and 上一个掩码即可。
  • 如果阶码等于零，整体右移，再处理一下符号位即可，讨论小数位首位即可确定正确性。

121212 步。

unsigned float_twice(unsigned uf) {
    unsigned t=(uf&2139095040)>>23;
    if(t==255) return uf;
    if(t>0) {
        uf+=1<<23;
        if(t==254) uf&=4286578688;
    }
    else {
        unsigned FL=uf&(1<<31);
        uf<<=1,uf+=FL;
    }
    return uf;
}

float_f2i

讨论的，很难受。
首先取出符号位和阶码 EEE，分类讨论：

  • E≤−1E\le -1E≤−1，绝对值一定小于 111，向零舍入到 000。
  • E≥31E\ge 31E≥31，绝对值一定大于 2322^{32}232，溢出或者本身就是 −231-2^{31}−231。
  • E<23E<23E<23，此时存在小数部分，需要把小数部分截去，构造一个掩码仅涵盖小数位前 EEE 位，记得加上 2E2^E2E。
  • E≥23E\ge 23E≥23，此时不存在小数，直接取出小数部分右移动若干位，记得加上 2E2^E2E。

int float_f2i(unsigned uf) {
    int FL,E,tmp,fr;
    FL=uf&(1<<31),E=((uf&2139095040)>>23)-127;
    if(E<=-1) return 0;
    if(E>=31) return 0x80000000u;
    if(E<23) {
        tmp=~(-8388609+(1<<(23-E)));
        fr=((uf&tmp)>>(23-E))+(1<<E);
    }
    else fr=((uf&8388607)<<(E-23))+(1<<E);
    if(FL) fr=(~fr)+1;
    return fr;
}

float_negpwr2

比上面两个简单太多了。
为了方便讨论，首先对 xxx 取反，注意这里可能会被 −231-2^{31}−231 阴一手哈哈。

  • 计算 Float 的最大表示范围可以知道 x∈[−149,127]x\in[-149,127]x∈[−149,127]，大了小了都可以简单地判掉。
  • 规格化数相当于 x≥−126x\ge -126x≥−126，只需要填好阶码就行。
  • 非规格化数相当于填小数位，这个直接填 222 的若干次幂即可。

121212 步。

unsigned float_negpwr2(int x) {
    if(x==-2147483648) return 2139095040;
    x=-x;
    if(x<-149) return 0;
    if(x>127) return 2139095040;
    if(x>=-126) return (x+127)<<23;
    return 1<<(x+149);
}

float_greater

拆出符号位、阶码、小数位，一个一个比较即可。
注意判掉 NaN 和 ±0.0\pm0.0±0.0，不算困难，313131 步。

unsigned float_greater(unsigned x, unsigned y) {
    unsigned Fx,Ex,Fy,Ey,fx,fy;
    Fx=x&(1u<<31),Ex=((x&2139095040)>>23),fx=x&8388607;
    Fy=y&(1u<<31),Ey=((y&2139095040)>>23),fy=y&8388607;
    if(Ex==255&&fx) return 0;
    if(Ey==255&&fy) return 0;
    if(Ex==0&&Ey==0&&fx==0&&fy==0) return 0;
    if(Fx<Fy) return 1;
    else if(Fx==Fy) {
        if(Fx) {
            if(Ex<Ey) return 1;
            else if(Ex==Ey&&fx<fy) return 1;
        }
        else {
            if(Ex>Ey) return 1;
            else if(Ex==Ey&&fx>fy) return 1;
        }
    }
    return 0;
}
